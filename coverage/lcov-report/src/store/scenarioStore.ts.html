import { create } from 'zustand';
import { persist, subscribeWithSelector } from 'zustand/middleware';
import type { Scenario, ScenarioState } from '../types';
import { masterStorage } from '../utils/masterStorage';
 
/**
 * Zustand store managing budgeting/forecasting scenarios, the currently selected
 * scenario, the active comparison set, and transient UI flags (loading, error).
 * State is persisted under the `scenario-store` key via `masterStorage`.
 */
export const useScenarioStore = create<ScenarioState>()(
  subscribeWithSelector(
    persist(
      (set) => ({
        scenarios: [],
        selectedScenarioId: null,
        comparedScenarioIds: [],
        isLoading: false,
        error: null,
 
        /**
         * Sets the global error message for the scenarios slice.
         * @param error - Error string to surface, or `null` to clear it.
         */
        setError: (error) => set({ error }),
 
        /** Clears the current error message, setting it back to `null`. */
        clearError: () => set({ error: null }),
 
        /**
         * Toggles the loading flag used by the UI to show spinners/skeletons.
         * @param loading - `true` while a scenario operation is in flight.
         */
        setLoading: (loading) => set({ isLoading: loading }),
 
        /**
         * Replaces the entire scenarios collection (e.g. after a bulk fetch).
         * @param scenarios - The new list of scenarios to store.
         */
        setScenarios: (scenarios) => set({ scenarios }),
 
        /**
         * Selects a scenario by id, driving detail views and reports.
         * @param id - Scenario id to mark as selected, or `null` to deselect.
         */
        setSelectedScenario: (id) => set({ selectedScenarioId: id }),
 
        /**
         * Creates a new scenario, generating its id and timestamps, appends it
         * to the store, and returns the new id.
         * @param scenario - Partial scenario payload; `id`, `createdAt`, and
         *   `updatedAt` are filled in automatically.
         * @returns The generated scenario id.
         */
        createScenario: (scenario) => {
          const newScenario: Scenario = {
            ...scenario,
            id: `scn-${Date.now()}`,
            createdAt: new Date().toISOString(),
            updatedAt: new Date().toISOString(),
          };
          set((state) => ({ scenarios: [...state.scenarios, newScenario] }));
          return newScenario.id;
        },
 
        /**
         * Applies a partial update to an existing scenario and refreshes its
         * `updatedAt` timestamp. No-ops silently if the id is not found.
         * @param id - Id of the scenario to update.
         * @param updates - Partial fields to merge over the existing scenario.
         */
        updateScenario: (id, updates) => {
          set((state) => ({
            scenarios: state.scenarios.map((s) =>
              s.id === id ? { ...s, ...updates, updatedAt: new Date().toISOString() } : s
            ),
          }));
        },
 
        /**
         * Removes a scenario from the store. If the deleted scenario was
         * currently selected, the selection is reset to `null`.
         * @param id - Id of the scenario to remove.
         */
        deleteScenario: (id) => {
          set((state) => ({
            scenarios: state.scenarios.filter((s) => s.id !== id),
            selectedScenarioId: state.selectedScenarioId === id ? null : state.selectedScenarioId,
          }));
        },
 
        /**
         * Toggles a scenario's membership in the comparison set used by
         * side-by-side reports. Adds the id if absent, removes it if present.
         * @param id - Id of the scenario to toggle in the comparison list.
         */
        toggleScenarioComparison: (id) => {
          set((state) => {
            const isCompared = state.comparedScenarioIds.includes(id);
            return {
              comparedScenarioIds: isCompared
                ? state.comparedScenarioIds.filter((cid) => cid !== id)
                : [...state.comparedScenarioIds, id],
            };
          });
        },
      }),
      {
        name: 'scenario-store',
        storage: masterStorage,
      }
    )
  )
);
 
/**
 * Memoized Zustand selectors for `useScenarioStore`. Using these in components
 * (via `useScenarioStore(scenarioSelectors.x)`) ensures re-renders only fire
 * when the selected slice actually changes.
 */
export const scenarioSelectors = {
  /** Returns the full list of stored scenarios. */
  scenarios: (state: ScenarioState) => state.scenarios,
  /** Returns the id of the currently selected scenario, or `null`. */
  selectedScenarioId: (state: ScenarioState) => state.selectedScenarioId,
  /** Returns the ids of scenarios currently marked for comparison. */
  comparedScenarioIds: (state: ScenarioState) => state.comparedScenarioIds,
  /** Returns whether a scenario operation is currently in flight. */
  isLoading: (state: ScenarioState) => state.isLoading,
  /** Derived: total number of stored scenarios. */
  scenarioCount: (state: ScenarioState) => state.scenarios.length,
  /** Derived: the selected scenario object, or `null` if none/invalid. */
  selectedScenario: (state: ScenarioState) =>
    state.scenarios.find((s) => s.id === state.selectedScenarioId) ?? null,
  /** Derived: the scenario objects currently in the comparison set. */
  comparedScenarios: (state: ScenarioState) =>
    state.scenarios.filter((s) => state.comparedScenarioIds.includes(s.id)),
  /** Derived: `true` when at least one scenario exists in the store. */
  hasScenarios: (state: ScenarioState) => state.scenarios.length > 0,
  /** Derived: `true` when the comparison set is non-empty. */
  isComparing: (state: ScenarioState) => state.comparedScenarioIds.length > 0,
};
