/**
 * Aggregate Table Engine — Pre-aggregate data at different granularity levels
 */
 
/** Time-bucketing strategy applied to source periods. */
export type Granularity = 'monthly' | 'quarterly' | 'yearly' | 'ytd';
 
export interface AggregateKey {
  entityId: string;
  accountId: string;
  period: string;
  granularity: Granularity;
}
 
export interface AggregateEntry {
  key: AggregateKey;
  debit: number;
  credit: number;
  net: number;
  count: number;
  lastUpdated: number;
}
 
export interface AggregateInputRow {
  entityId: string;
  accountId: string;
  period: string;
  debit: number;
  credit: number;
}
 
/**
 * In-memory store of pre-aggregated ledger entries keyed by entity, account,
 * rolled-up period, and granularity. Use the static API to ingest rows,
 * perform point/range lookups, invalidate subsets of the cache, and report
 * memory statistics.
 */
export class AggregateTableEngine {
  private static aggregates = new Map<string, AggregateEntry>();
  private static dirtyKeys = new Set<string>();
 
  private static makeKey(key: AggregateKey): string {
    return `${key.entityId}:${key.accountId}:${key.period}:${key.granularity}`;
  }
 
  /**
   * Ingest source rows and merge their debit/credit amounts into existing
   * aggregates for the rolled-up period. New combinations create entries;
   * existing ones are incremented in place. All touched keys are marked
   * dirty for downstream cache subscribers.
   *
   * @param data Source rows to aggregate. Periods are rolled up to the
   *   provided `granularity` (e.g. `2024-03` becomes `2024-Q1` for
   *   `quarterly`).
   * @param granularity Time bucket to roll rows into.
   */
  static aggregate(data: AggregateInputRow[], granularity: Granularity): void {
    for (const row of data) {
      const aggPeriod = this.rollupPeriod(row.period, granularity);
      const key: AggregateKey = {
        entityId: row.entityId,
        accountId: row.accountId,
        period: aggPeriod,
        granularity,
      };
      const mapKey = this.makeKey(key);
      const existing = this.aggregates.get(mapKey);
 
      this.dirtyKeys.add(mapKey);
 
      if (existing) {
        existing.debit += row.debit;
        existing.credit += row.credit;
        existing.net = existing.debit - existing.credit;
        existing.count++;
        existing.lastUpdated = Date.now();
      } else {
        this.aggregates.set(mapKey, {
          key,
          debit: row.debit,
          credit: row.credit,
          net: row.debit - row.credit,
          count: 1,
          lastUpdated: Date.now(),
        });
      }
    }
  }
 
  /**
   * Look up a single aggregate entry by its full key. The `period` argument
   * must already be expressed at the target `granularity`.
   *
   * @param entityId Entity identifier (legal entity / business unit).
   * @param accountId Account identifier within the entity.
   * @param period Period label at the requested granularity
   *   (e.g. `2024-07` for `monthly`, `2024-Q3` for `quarterly`).
   * @param granularity Granularity the period is expressed in.
   * @returns The matching aggregate entry, or `undefined` when no data
   *   has been aggregated for that key.
   */
  static query(
    entityId: string,
    accountId: string,
    period: string,
    granularity: Granularity
  ): AggregateEntry | undefined {
    const key: AggregateKey = { entityId, accountId, period, granularity };
    return this.aggregates.get(this.makeKey(key));
  }
 
  /**
   * Return all aggregate entries for an entity/account pair whose period
   * falls within the inclusive `[startPeriod, endPeriod]` range at the
   * given granularity. Periods are compared lexicographically, so the
   * caller must pass values produced by the same formatting convention
   * used during ingestion.
   *
   * @param entityId Entity identifier.
   * @param accountId Account identifier.
   * @param startPeriod Inclusive lower bound of the period range.
   * @param endPeriod Inclusive upper bound of the period range.
   * @param granularity Granularity shared by both bounds and the stored
   *   entries.
   */
  static queryRange(
    entityId: string,
    accountId: string,
    startPeriod: string,
    endPeriod: string,
    granularity: Granularity
  ): AggregateEntry[] {
    return Array.from(this.aggregates.values()).filter(
      (e) =>
        e.key.entityId === entityId &&
        e.key.accountId === accountId &&
        e.key.granularity === granularity &&
        e.key.period >= startPeriod &&
        e.key.period <= endPeriod
    );
  }
 
  /**
   * Drop aggregate entries from the cache. With no arguments the entire
   * table is cleared. Passing `entityId` and/or `accountId` scopes the
   * eviction to matching keys; deleted keys are added to the dirty set so
   * subscribers can refetch them.
   *
   * @param entityId Optional entity to invalidate. Omit to match all.
   * @param accountId Optional account to invalidate. Omit to match all.
   */
  static invalidate(entityId?: string, accountId?: string): void {
    if (!entityId && !accountId) {
      this.aggregates.clear();
      this.dirtyKeys.clear();
      return;
    }
    for (const [key, entry] of this.aggregates) {
      if (entityId && entry.key.entityId !== entityId) continue;
      if (accountId && entry.key.accountId !== accountId) continue;
      this.aggregates.delete(key);
      this.dirtyKeys.add(key);
    }
  }
 
  /**
   * Snapshot of cache usage metrics: total entries held, the number of
   * keys currently flagged dirty, and a rough memory estimate in
   * kilobytes (assumes ~128 bytes per entry).
   */
  static getStats(): { totalEntries: number; dirtyKeys: number; memoryEstimateKB: number } {
    return {
      totalEntries: this.aggregates.size,
      dirtyKeys: this.dirtyKeys.size,
      memoryEstimateKB: Math.round((this.aggregates.size * 128) / 1024),
    };
  }
 
  private static rollupPeriod(period: string, granularity: Granularity): string {
    if (!period || granularity === 'monthly') return period;
 
    // Support YYYY-MM and YYYY formats
    const year = period.slice(0, 4);
    const monthPart = period.slice(5, 7);
    const month = parseInt(monthPart, 10) || 1;
 
    switch (granularity) {
      case 'quarterly': {
        const q = Math.ceil(month / 3);
        return `${year}-Q${q}`;
      }
      case 'yearly':
        return year;
      case 'ytd':
        return `${year}-YTD`;
      default:
        return period;
    }
  }
}
