// =============================================================================
// AGGREGATION DESIGNER — Pre-aggregate common queries for speed
// Creates materialized aggregation tables for frequently-queried slices
// Pure TypeScript, deterministic, testable, zero external dependencies
// =============================================================================
 
export interface AggregationDefinition {
  id: string;
  name: string;
  cube: string;
  dimensions: string[];
  measures: string[];
  aggregation: 'sum' | 'avg' | 'count' | 'min' | 'max';
  materialized: boolean;
  createdAt: string;
  updatedAt: string;
  cellCount: number;
}
 
export interface AggregationCell {
  key: string;
  dimensions: Record<string, string>;
  measure: string;
  value: number;
  count: number;
}
 
export interface AggregationResult {
  definitionId: string;
  cells: AggregationCell[];
  queryTimeMs: number;
  hitCount: number;
}
 
// =============================================================================
// AGGREGATION DESIGNER
// =============================================================================
 
/**
 * Designer for pre-aggregated, materialized query results over OLAP-style cubes.
 *
 * `AggregationDesigner` lets callers declare aggregation definitions (a cube,
 * a set of dimensions, a set of measures, and a reduction function) and then
 * materialize them from raw cell data so subsequent queries can be served from
 * an in-memory map. It is intentionally pure-TypeScript, deterministic, and
 * dependency-free so it can run in workers, tests, or restricted sandboxes.
 *
 * @example
 * ```ts
 * const designer = new AggregationDesigner();
 * const def = designer.define('Revenue by Region', 'sales', ['region'], ['amount'], 'sum');
 * designer.materialize(def.id, rawCells);
 * const result = designer.query(def.id, { region: 'EMEA' });
 * ```
 */
export class AggregationDesigner {
  private definitions = new Map<string, AggregationDefinition>();
  private aggregatedData = new Map<string, Map<string, AggregationCell>>();
  private hitCounts = new Map<string, number>();
 
  /**
   * Register a new aggregation definition.
   *
   * Generates a unique id, sorts the dimension list for stable key generation,
   * and stores the definition in an unsynchronized `Map`. The definition is
   * not materialized yet — call `materialize` to populate the cell store.
   *
   * @param name - Human-readable label for the aggregation.
   * @param cube - Logical cube name the definition is bound to. Only cells
   *   whose key includes `cube=<this value>` will be aggregated.
   * @param dimensions - Dimension names to group by. Order is normalized via
   *   a sort so that equivalent definitions produce identical keys.
   * @param measures - Measure names this definition reduces over. Cells whose
   *   `measure=` segment is not in this set are skipped during materialization.
   * @param aggregation - Reduction function applied per cell group. Defaults
   *   to `'sum'`.
   * @returns The newly created {@link AggregationDefinition}, including its
   *   generated id and timestamps.
   */
  define(
    name: string,
    cube: string,
    dimensions: string[],
    measures: string[],
    aggregation: 'sum' | 'avg' | 'count' | 'min' | 'max' = 'sum'
  ): AggregationDefinition {
    const id = `agg-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
    const def: AggregationDefinition = {
      id,
      name,
      cube,
      dimensions: dimensions.slice().sort(),
      measures,
      aggregation,
      materialized: false,
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString(),
      cellCount: 0,
    };
    this.definitions.set(id, def);
    return def;
  }
 
  /**
   * Look up a single aggregation definition by id.
   *
   * @param id - Definition id returned from {@link define}.
   * @returns The matching {@link AggregationDefinition}, or `undefined` if no
   *   definition exists for the given id.
   */
  getDefinition(id: string): AggregationDefinition | undefined {
    return this.definitions.get(id);
  }
 
  /**
   * List every aggregation definition currently registered, in insertion
   * order (the underlying `Map` preserves insertion order).
   *
   * @returns A new array containing all definitions. The array is a fresh
   *   snapshot; mutating it does not affect internal state.
   */
  listDefinitions(): AggregationDefinition[] {
    return Array.from(this.definitions.values());
  }
 
  /**
   * Remove a definition and any of its materialized state.
   *
   * Also clears the associated aggregated-data map and hit counter so the
   * id can be safely reused (or just to release memory).
   *
   * @param id - Definition id to remove.
   * @returns `true` if a definition existed and was deleted; `false` if the
   *   id was unknown.
   */
  deleteDefinition(id: string): boolean {
    this.aggregatedData.delete(id);
    this.hitCounts.delete(id);
    return this.definitions.delete(id);
  }
 
  /**
   * Materialize an aggregation definition from raw cell data.
   *
   * Iterates the input map, parses each key (expected to be a `|`-separated
   * list of `name=value` segments), filters to entries whose `cube=` segment
   * matches the definition and whose `measure=` segment is in the definition's
   * measure set, groups them by the configured dimensions, and applies the
   * configured reduction (`sum`/`avg`/`count`/`min`/`max`). For `'avg'`, the
   * running sum is divided by the running count after the pass.
   *
   * On success the definition's `materialized`, `cellCount`, and `updatedAt`
   * fields are refreshed.
   *
   * @param definitionId - Id of a previously {@link define}d aggregation.
   * @param cells - Raw cell map keyed by `|`-separated coordinate strings.
   *   Non-numeric values are coerced to `0` for the reduction step.
   * @returns The updated definition, or `null` if `definitionId` is unknown.
   */
  materialize(definitionId: string, cells: Map<string, unknown>): AggregationDefinition | null {
    const def = this.definitions.get(definitionId);
    if (!def) return null;
 
    const aggregated = new Map<string, AggregationCell>();
    const dimSet = new Set(def.dimensions);
    const measureSet = new Set(def.measures);
 
    for (const [key, value] of cells) {
      let matchesCube = false;
      let measureKey = '';
      const dimVals: Record<string, string> = {};
 
      const parts = key.split('|');
      for (let i = 0; i < parts.length; i++) {
        const part = parts[i];
        const eqIdx = part.indexOf('=');
        if (eqIdx > 0) {
          const k = part.slice(0, eqIdx);
          const v = part.slice(eqIdx + 1);
          if (k === 'cube' && v === def.cube) {
            matchesCube = true;
          } else if (k === 'measure') {
            measureKey = v;
          } else if (dimSet.has(k)) {
            dimVals[k] = v;
          }
        }
      }
 
      if (!matchesCube || !measureSet.has(measureKey)) continue;
 
      let aggKey = '';
      let isFirst = true;
      for (let i = 0; i < def.dimensions.length; i++) {
        const d = def.dimensions[i];
        const v = dimVals[d];
        if (v !== undefined) {
          if (!isFirst) aggKey += '|';
          aggKey += `${d}=${v}`;
          isFirst = false;
        }
      }
 
      const fullKey = aggKey ? `${aggKey}|${measureKey}` : measureKey;
      const numValue = typeof value === 'number' ? value : 0;
 
      let cell = aggregated.get(fullKey);
      if (!cell) {
        cell = {
          key: fullKey,
          dimensions: dimVals,
          measure: measureKey,
          value: 0,
          count: 0,
        };
        aggregated.set(fullKey, cell);
      }
 
      switch (def.aggregation) {
        case 'sum':
          cell.value += numValue;
          break;
        case 'avg':
          cell.value += numValue;
          cell.count++;
          break;
        case 'count':
          cell.count++;
          cell.value = cell.count;
          break;
        case 'min':
          if (cell.count === 0) cell.value = numValue;
          else cell.value = Math.min(cell.value, numValue);
          cell.count++;
          break;
        case 'max':
          if (cell.count === 0) cell.value = numValue;
          else cell.value = Math.max(cell.value, numValue);
          cell.count++;
          break;
      }
    }
 
    if (def.aggregation === 'avg') {
      for (const cell of aggregated.values()) {
        if (cell.count > 0) {
          cell.value = cell.value / cell.count;
        }
      }
    }
 
    this.aggregatedData.set(definitionId, aggregated);
    def.materialized = true;
    def.cellCount = aggregated.size;
    def.updatedAt = new Date().toISOString();
    return def;
  }
 
  /**
   * Query a materialized aggregation, optionally filtering by dimension
   * values.
   *
   * Measures the elapsed time in milliseconds using `performance.now()` and
   * increments the per-definition hit counter, which is returned in the
   * result so callers can monitor cache effectiveness.
   *
   * @param definitionId - Id of a previously {@link define}d aggregation.
   * @param filters - Optional dimension equality filters. A cell matches if
   *   every entry's dimension value equals the filter value. Omit to return
   *   all materialized cells.
   * @returns An {@link AggregationResult} with the matched cells, query time,
   *   and cumulative hit count. Returns `null` if the definition is unknown,
   *   and an empty result (with `queryTimeMs: 0`) if the definition exists
   *   but has not been materialized yet.
   */
  query(definitionId: string, filters?: Record<string, string>): AggregationResult | null {
    const def = this.definitions.get(definitionId);
    if (!def) return null;
 
    const data = this.aggregatedData.get(definitionId);
    if (!data) return { definitionId, cells: [], queryTimeMs: 0, hitCount: 0 };
 
    const start = performance.now();
    const cells: AggregationCell[] = [];
 
    if (filters) {
      const filterEntries = Object.entries(filters);
      const filterLen = filterEntries.length;
      for (const cell of data.values()) {
        let match = true;
        for (let i = 0; i < filterLen; i++) {
          if (cell.dimensions[filterEntries[i][0]] !== filterEntries[i][1]) {
            match = false;
            break;
          }
        }
        if (match) cells.push(cell);
      }
    } else {
      for (const cell of data.values()) {
        cells.push(cell);
      }
    }
 
    const hitCount = (this.hitCounts.get(definitionId) ?? 0) + 1;
    this.hitCounts.set(definitionId, hitCount);
 
    return {
      definitionId,
      cells,
      queryTimeMs: performance.now() - start,
      hitCount,
    };
  }
 
  /**
   * Re-run materialization for an existing definition using fresh cell data.
   *
   * Convenience wrapper that delegates to {@link materialize}. Useful when
   * underlying cube data has changed and the materialized snapshot needs to
   * be rebuilt.
   *
   * @param definitionId - Id of a previously {@link define}d aggregation.
   * @param cells - New raw cell map (same shape as in `materialize`).
   * @returns `true` if the definition existed and was re-materialized;
   *   `false` if the id is unknown.
   */
  refresh(definitionId: string, cells: Map<string, unknown>): boolean {
    const def = this.definitions.get(definitionId);
    if (!def) return false;
    this.materialize(definitionId, cells);
    return true;
  }
 
  /**
   * Compute aggregate statistics across all registered definitions.
   *
   * @returns An object containing:
   *   - `totalDefinitions`: count of definitions currently registered.
   *   - `materializedCount`: subset of definitions that have been
   *     materialized at least once.
   *   - `totalCells`: sum of `cellCount` across all definitions — the
   *     total number of materialized cells in memory.
   */
  getStats(): { totalDefinitions: number; materializedCount: number; totalCells: number } {
    const defs = Array.from(this.definitions.values());
    return {
      totalDefinitions: defs.length,
      materializedCount: defs.filter((d) => d.materialized).length,
      totalCells: defs.reduce((sum, d) => sum + d.cellCount, 0),
    };
  }
 
  private parseCellKey(key: string): Record<string, string> {
    const coords: Record<string, string> = {};
    const parts = key.split('|');
    for (const part of parts) {
      const eqIdx = part.indexOf('=');
      if (eqIdx > 0) coords[part.slice(0, eqIdx)] = part.slice(eqIdx + 1);
    }
    return coords;
  }
}
