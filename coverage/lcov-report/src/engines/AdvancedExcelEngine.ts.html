/**
 * AdvancedExcelEngine — Enhanced Excel export for FinPlan Pro
 * Features: conditional formatting, named ranges, multi-sheet, cell comments
 */
 
/**
 * Sheet definition consumed by `AdvancedExcelEngine`.
 *
 * @template T Type of each cell value. Defaults to `unknown` for general use.
 */
export interface ExcelSheet<T = unknown> {
  name: string;
  data: T[][];
  columns: Array<{ header: string; width: number }>;
  conditionalFormatting?: ConditionalFormat[];
  comments?: CellComment[];
  namedRanges?: NamedRange[];
  freezeRows?: number;
  freezeCols?: number;
}
 
/**
 * Conditional-formatting rule applied to a sheet range.
 *
 * `value` interpretation depends on `type`:
 * - `'between'` expects a two-element array as a string pair in `value`.
 * - `'text'` expects a literal substring.
 * - numeric types compare against the numeric `value`.
 */
export interface ConditionalFormat {
  range: string;
  type: 'greaterThan' | 'lessThan' | 'between' | 'text' | 'custom';
  value: number | string;
  style: { backgroundColor?: string; fontColor?: string; bold?: boolean };
}
 
/**
 * Cell-level comment attached to a sheet.
 */
export interface CellComment {
  cell: string;
  author: string;
  text: string;
  timestamp: string;
}
 
/**
 * Workbook-scoped named range descriptor.
 */
export interface NamedRange {
  name: string;
  range: string;
  sheet: string;
}
 
/**
 * Workbook-level metadata passed to `AdvancedExcelEngine.createWorkbook`.
 */
export interface ExcelExportOptions {
  password?: string;
  author?: string;
  title?: string;
  subject?: string;
  freezePanes?: { rows: number; cols: number }[];
}
 
export class AdvancedExcelEngine {
  /**
   * Build a multi-sheet XLSX-compatible workbook blob from the provided sheets.
   *
   * @template T Row payload type carried by each sheet.
   * @param sheets Ordered list of sheets to include in the workbook.
   * @param options Optional workbook-level metadata (title, author, password, freeze panes).
   * @returns A `Blob` containing the generated XML in `application/vnd.openxmlformats-officedocument.spreadsheetml.sheet` MIME type.
   */
  static createWorkbook<T>(sheets: ExcelSheet<T>[], options?: ExcelExportOptions): Blob {
    // Create XLSX-compatible XML
    const xml = this.generateXLSX(sheets, options);
    return new Blob([xml], {
      type: 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
    });
  }
 
  /**
   * Generate XLSX XML content
   */
  private static generateXLSX<T>(sheets: ExcelSheet<T>[], options?: ExcelExportOptions): string {
    const sheetCount = sheets.length;
    let xml = '<?xml version="1.0" encoding="UTF-8" standalone="yes"?>\n';
    xml += '<workbook xmlns="http://schemas.openxmlformats.org/spreadsheetml/2006/main"\n';
    xml += '  xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships">\n';
    xml += '  <sheets>\n';
 
    for (let i = 0; i < sheetCount; i++) {
      xml += `    <sheet name="${sheets[i].name}" sheetId="${i + 1}" r:id="rId${i + 1}" />\n`;
    }
 
    xml += '  </sheets>\n';
    xml += '</workbook>';
 
    return xml;
  }
 
  /**
   * Export a flat array of records to a single-sheet XLSX blob.
   *
   * Missing keys fall back to an empty string so the output always has a complete
   * rectangular grid of cells.
   *
   * @template T Row record type; must be an object with string keys.
   * @param data Source records to serialize as rows.
   * @param columns Column descriptors pairing a header label with the record key to project.
   * @param sheetName Name of the generated sheet. Defaults to `'Sheet1'`.
   * @returns A `Blob` containing the single-sheet workbook.
   */
  static exportToExcel<T extends Record<string, unknown>>(
    data: T[],
    columns: Array<{ header: string; key: Extract<keyof T, string>; width?: number }>,
    sheetName: string = 'Sheet1'
  ): Blob {
    const colCount = columns.length;
    const rowCount = data.length;
 
    const headers = new Array<string>(colCount);
    const cols = new Array<{ header: string; width: number }>(colCount);
    for (let i = 0; i < colCount; i++) {
      const c = columns[i];
      headers[i] = c.header;
      cols[i] = { header: c.header, width: c.width ?? 150 };
    }
 
    const rows = new Array<unknown[]>(rowCount + 1);
    rows[0] = headers;
 
    for (let i = 0; i < rowCount; i++) {
      const rowData = new Array<unknown>(colCount);
      const row = data[i];
      for (let j = 0; j < colCount; j++) {
        rowData[j] = row[columns[j].key] ?? '';
      }
      rows[i + 1] = rowData;
    }
 
    const sheet: ExcelSheet<unknown> = {
      name: sheetName,
      data: rows,
      columns: cols,
    };
 
    return this.createWorkbook([sheet]);
  }
 
  /**
   * Export a multi-section financial report as a multi-sheet workbook.
   *
   * Each section becomes its own sheet named after `section.name`, and the
   * workbook metadata is seeded from `report.title`.
   *
   * @param report Report definition containing a title and an ordered list of sections.
   * @returns A `Blob` containing the generated multi-sheet workbook.
   */
  static exportFinancialReport(report: {
    title: string;
    sections: Array<{
      name: string;
      data: Record<string, unknown>[];
      columns: Array<{ header: string; key: string }>;
    }>;
  }): Blob {
    const sectionCount = report.sections.length;
    const sheets = new Array<ExcelSheet<unknown>>(sectionCount);
 
    for (let s = 0; s < sectionCount; s++) {
      const section = report.sections[s];
      const colCount = section.columns.length;
      const dataCount = section.data.length;
 
      const headers = new Array<string>(colCount);
      const cols = new Array<{ header: string; width: number }>(colCount);
      for (let i = 0; i < colCount; i++) {
        const c = section.columns[i];
        headers[i] = c.header;
        cols[i] = { header: c.header, width: 150 };
      }
 
      const rows = new Array<unknown[]>(dataCount + 1);
      rows[0] = headers;
 
      for (let i = 0; i < dataCount; i++) {
        const rowData = new Array<unknown>(colCount);
        const row = section.data[i];
        for (let j = 0; j < colCount; j++) {
          rowData[j] = row[section.columns[j].key] ?? '';
        }
        rows[i + 1] = rowData;
      }
 
      sheets[s] = {
        name: section.name,
        data: rows,
        columns: cols,
      };
    }
 
    return this.createWorkbook(sheets, { title: report.title });
  }
 
  /**
   * Return a new sheet with the provided conditional-format rules merged in.
   *
   * Existing rules on the input sheet (if any) are preserved; the new rules
   * are appended. The input sheet is not mutated.
   *
   * @template T Row payload type carried by the sheet.
   * @param sheet Sheet to extend.
   * @param formats Conditional-format rules to append.
   * @returns A new `ExcelSheet` instance with the merged rules.
   */
  static applyConditionalFormatting<T>(
    sheet: ExcelSheet<T>,
    formats: ConditionalFormat[]
  ): ExcelSheet<T> {
    return {
      ...sheet,
      conditionalFormatting: sheet.conditionalFormatting
        ? sheet.conditionalFormatting.concat(formats)
        : formats.slice(),
    };
  }
 
  /**
   * Return a new sheet with the provided cell comments merged in.
   *
   * Existing comments on the input sheet (if any) are preserved; the new
   * comments are appended. The input sheet is not mutated.
   *
   * @template T Row payload type carried by the sheet.
   * @param sheet Sheet to extend.
   * @param comments Cell comments to append.
   * @returns A new `ExcelSheet` instance with the merged comments.
   */
  static addComments<T>(sheet: ExcelSheet<T>, comments: CellComment[]): ExcelSheet<T> {
    return {
      ...sheet,
      comments: sheet.comments ? sheet.comments.concat(comments) : comments.slice(),
    };
  }
 
  /**
   * Build a `NamedRange` descriptor from its three required components.
   *
   * @param name Friendly name to expose in the workbook.
   * @param range A1-style cell range, e.g. `'A1:D10'`.
   * @param sheet Sheet the range belongs to.
   * @returns A `NamedRange` value object.
   */
  static createNamedRange(name: string, range: string, sheet: string): NamedRange {
    return { name, range, sheet };
  }
 
  /**
   * Parse a tab-delimited, UTF-8 encoded buffer into a single sheet.
   *
   * This is a simplified import path: rows are split on `\n` and cells on
   * `\t`. A real XLSX parser should be used for production imports.
   *
   * @param buffer Raw bytes of the tab-delimited file.
   * @returns A single-sheet workbook named `'Imported'` with the decoded rows.
   */
  static parseExcel(buffer: ArrayBuffer): ExcelSheet<string>[] {
    const decoder = new TextDecoder();
    const content = decoder.decode(buffer);
    const lines = content.split('\n');
    const rowCount = lines.length;
 
    const data = new Array<string[]>(rowCount);
    for (let i = 0; i < rowCount; i++) {
      data[i] = lines[i].split('\t');
    }
 
    return [
      {
        name: 'Imported',
        data,
        columns: [],
      },
    ];
  }
}
