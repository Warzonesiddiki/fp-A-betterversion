/**
 * AdvancedPDFEngine — Professional PDF generation for FinPlan Pro
 * Features: TOC, watermarks, headers/footers, charts, financial tables
 */
 
import jsPDF from 'jspdf';
 
interface TOCEntry {
  title: string;
  level: number;
  page: number;
}
 
interface WatermarkOptions {
  text: string;
  fontSize?: number;
  color?: string;
  opacity?: number;
  angle?: number;
}
 
interface HeaderFooterOptions {
  header?: { title: string; subtitle?: string };
  footer?: { showPageNumbers?: boolean; showDate?: boolean; text?: string };
}
 
interface FinancialTableColumn {
  key: string;
  header: string;
  width: number;
  align?: 'left' | 'right' | 'center';
  format?: 'currency' | 'number' | 'percent' | 'text';
}
 
export class AdvancedPDFEngine {
  /**
   * Render a table of contents page into the PDF document.
   *
   * Writes a centered "Table of Contents" heading followed by an indented list
   * of section titles and their target page numbers. Entries whose cumulative
   * Y position exceeds the page bottom trigger an automatic page break so the
   * TOC continues on a fresh page. The function always ends by inserting an
   * additional blank page so subsequent content starts on its own page.
   *
   * @param doc - The active jsPDF instance to render into.
   * @param sections - Ordered list of TOC entries to display. The `level` field
   * controls horizontal indentation (20px per level) and `page` is printed on
   * the right side of the row.
   * @returns void
   */
  static addTableOfContents(doc: jsPDF, sections: TOCEntry[]): void {
    doc.setFontSize(24);
    doc.text('Table of Contents', 20, 30);
    doc.setFontSize(12);
 
    let y = 50;
    for (const section of sections) {
      const indent = section.level * 20;
      doc.text(section.title, 20 + indent, y);
      doc.text(`${section.page}`, 180, y);
      y += 8;
      if (y > 270) {
        doc.addPage();
        y = 20;
      }
    }
    doc.addPage();
  }
 
  /**
   * Stamp a diagonal watermark text on every page of the document.
   *
   * Iterates over all existing pages, switches to each one, saves the
   * graphics state, and writes the watermark centered using the requested
   * font size, color, opacity, and rotation angle. The graphics state is
   * restored on completion so subsequent drawing operations are unaffected.
   *
   * @param doc - The active jsPDF instance to stamp.
   * @param options - Watermark configuration. All visual properties
   * (`fontSize`, `color`, `opacity`, `angle`) are optional and fall back to
   * sensible defaults (50pt, `#cccccc`, 0.3, -45°) when omitted.
   * @returns void
   */
  static addWatermark(doc: jsPDF, options: WatermarkOptions): void {
    const { text, fontSize = 50, color = '#cccccc', opacity = 0.3, angle = -45 } = options;
    const pages = doc.getNumberOfPages();
 
    for (let i = 1; i <= pages; i++) {
      doc.setPage(i);
      doc.saveGraphicsState();
      doc.setTextColor(color);
      doc.setFontSize(fontSize);
 
      const pageWidth = doc.internal.pageSize.getWidth();
      const pageHeight = doc.internal.pageSize.getHeight();
      const centerX = pageWidth / 2;
      const centerY = pageHeight / 2;
 
      doc.text(text, centerX, centerY, {
        align: 'center',
        angle: angle,
      });
 
      doc.restoreGraphicsState();
    }
  }
 
  /**
   * Apply a consistent header and/or footer to every page of the document.
   *
   * Walks every page and draws the configured header (title plus optional
   * subtitle with a horizontal divider line) at the top of the page, and/or
   * the footer (custom text, "Page X of Y", and an optional date stamp) at
   * the bottom. Either section can be omitted independently.
   *
   * @param doc - The active jsPDF instance to decorate.
   * @param options - Header/footer configuration. `header` and `footer` are
   * independently optional; within `footer`, `text`, `showPageNumbers`, and
   * `showDate` can be combined freely.
   * @returns void
   */
  static addHeaderFooter(doc: jsPDF, options: HeaderFooterOptions): void {
    const pages = doc.getNumberOfPages();
 
    for (let i = 1; i <= pages; i++) {
      doc.setPage(i);
      const pageWidth = doc.internal.pageSize.getWidth();
      const pageHeight = doc.internal.pageSize.getHeight();
 
      if (options.header) {
        doc.setFontSize(16);
        doc.text(options.header.title, 20, 15);
        if (options.header.subtitle) {
          doc.setFontSize(10);
          doc.text(options.header.subtitle, 20, 22);
        }
        doc.line(20, 25, pageWidth - 20, 25);
      }
 
      if (options.footer) {
        doc.setFontSize(8);
        doc.line(20, pageHeight - 15, pageWidth - 20, pageHeight - 15);
 
        if (options.footer.text) {
          doc.text(options.footer.text, 20, pageHeight - 10);
        }
 
        if (options.footer.showPageNumbers) {
          doc.text(`Page ${i} of ${pages}`, pageWidth - 40, pageHeight - 10);
        }
 
        if (options.footer.showDate) {
          doc.text(new Date().toLocaleDateString(), pageWidth / 2, pageHeight - 10);
        }
      }
    }
  }
 
  /**
   * Force a manual page break in the document.
   *
   * Thin convenience wrapper around `doc.addPage()` that documents intent
   * at call sites and provides a stable seam for future enhancements
   * (e.g. per-section page templates or running headers).
   *
   * @param doc - The active jsPDF instance to advance.
   * @returns void
   */
  static addPageBreak(doc: jsPDF): void {
    doc.addPage();
  }
 
  /**
   * Render a styled financial table with header row, alternating zebra
   * striping, and per-column value formatting.
   *
   * Draws a blue header band (white text), then iterates `data` rows drawing
   * each at `rowHeight` (20pt). Every other row receives a soft blue
   * background for legibility. Each column's `format` is applied at draw
   * time: `currency` adds a `$` prefix and 2-decimal grouping,
   * `percent` multiplies by 100 and appends `%`, `number` applies
   * locale grouping, and `text` (the default) stringifies the value. When
   * the cursor would overflow the page, a new page is started at Y=20.
   *
   * @param doc - The active jsPDF instance to draw the table into.
   * @param data - Array of row records. Each value is read via `row[col.key]`.
   * @param columns - Column descriptors defining `key`, `header`, `width`,
   * optional `align` (`'left'` | `'right'` | `'center'`) and `format`.
   * @param startY - Y coordinate (in mm) where the header row should start.
   * Defaults to `40` to leave room for a page header.
   * @returns The Y coordinate immediately after the last rendered row,
   * suitable for placing follow-up content (e.g. a chart caption).
   */
  static addFinancialTable(
    doc: jsPDF,
    data: Record<string, unknown>[],
    columns: FinancialTableColumn[],
    startY: number = 40
  ): number {
    let y = startY;
    const rowHeight = 20;
 
    // Header row
    doc.setFontSize(10);
    doc.setTextColor(255, 255, 255);
    doc.setFillColor(30, 64, 175);
    doc.rect(20, y, 170, rowHeight, 'F');
 
    let x = 22;
    for (const col of columns) {
      doc.text(col.header, x, y + 14);
      x += col.width;
    }
    y += rowHeight;
 
    // Data rows
    doc.setTextColor(30, 30, 30);
    for (let i = 0; i < data.length; i++) {
      const row = data[i];
      if (y > 270) {
        doc.addPage();
        y = 20;
      }
 
      if (i % 2 === 0) {
        doc.setFillColor(240, 245, 255);
        doc.rect(20, y, 170, rowHeight, 'F');
      }
 
      x = 22;
      for (const col of columns) {
        const value = row[col.key];
        let formatted: string;
 
        switch (col.format) {
          case 'currency':
            formatted =
              typeof value === 'number'
                ? `$${value.toLocaleString('en-US', { minimumFractionDigits: 2 })}`
                : String(value ?? '');
            break;
          case 'percent':
            formatted =
              typeof value === 'number' ? `${(value * 100).toFixed(1)}%` : String(value ?? '');
            break;
          case 'number':
            formatted =
              typeof value === 'number' ? value.toLocaleString('en-US') : String(value ?? '');
            break;
          default:
            formatted = String(value ?? '');
        }
 
        if (col.align === 'right') {
          doc.text(formatted, x + col.width - 4, y + 14, { align: 'right' });
        } else {
          doc.text(formatted, x, y + 14);
        }
        x += col.width;
      }
      y += rowHeight;
    }
 
    return y;
  }
 
  /**
   * Rasterize an inline SVG element and embed it as a PNG image in the PDF.
   *
   * Serializes the SVG to a data URL, draws it onto a 2x-scaled canvas with
   * a white background (so transparent SVGs render correctly in PDF), and
   * uses `doc.addImage` to place the resulting PNG at the requested
   * position and dimensions. The function is a no-op when `svgElement` is
   * `null` or when a 2D canvas context cannot be obtained.
   *
   * Note: image decoding is asynchronous via the `Image.onload` callback,
   * so callers should ensure the document is not saved before the next
   * animation frame to avoid races.
   *
   * @param doc - The active jsPDF instance to receive the image.
   * @param svgElement - The live `<svg>` DOM node to render, or `null` to skip.
   * @param x - Left X coordinate (mm) for the placed image. Defaults to `20`.
   * @param y - Top Y coordinate (mm) for the placed image. Defaults to `40`.
   * @param width - Target rendered width in mm. Defaults to `170`.
   * @param height - Target rendered height in mm. Defaults to `120`.
   * @returns void
   */
  static addChart(
    doc: jsPDF,
    svgElement: SVGElement | null,
    x: number = 20,
    y: number = 40,
    width: number = 170,
    height: number = 120
  ): void {
    if (!svgElement) return;
 
    const svgData = new XMLSerializer().serializeToString(svgElement);
    const canvas = document.createElement('canvas');
    const ctx = canvas.getContext('2d');
    if (!ctx) return;
 
    const img = new Image();
    img.onload = () => {
      canvas.width = img.width * 2;
      canvas.height = img.height * 2;
      ctx.scale(2, 2);
      ctx.fillStyle = '#ffffff';
      ctx.fillRect(0, 0, img.width, img.height);
      ctx.drawImage(img, 0, 0);
 
      const imgData = canvas.toDataURL('image/png');
      doc.addImage(imgData, 'PNG', x, y, width, height);
    };
    img.src = 'data:image/svg+xml;base64,' + btoa(unescape(encodeURIComponent(svgData)));
  }
 
  /**
   * Compose a complete financial report PDF from a section list.
   *
   * The pipeline is: (1) optional title page, (2) TOC placeholder page
   * bookmark, (3) one page per section containing the section title,
   * optional wrapped prose content, optional financial table, and optional
   * chart, (4) post-pass to fill the TOC with collected page numbers,
   * (5) header/footer applied to every page, and (6) optional diagonal
   * watermark over everything.
   *
   * @param sections - Ordered list of report sections. Each section may
   * carry `content` (long-form prose that auto-wraps at 170mm),
   * `table` (`{ data, columns }` passed through to `addFinancialTable`),
   * and `chart` (an SVG element passed through to `addChart`).
   * @param options - Optional report-level configuration:
   * - `title` / `subtitle` — printed on the cover page; `title` is also
   *   reused in the running header when present.
   * - `watermark` — text to overlay diagonally on every page.
   * - `includeTOC` — when true, renders a table of contents page from the
   *   collected section page numbers.
   * - `includePageNumbers` — reserved flag for caller intent; the engine
   *   always adds page numbers via the default footer when no `footer` is
   *   provided.
   * @returns A fully composed `jsPDF` document instance ready for preview
   * or `exportToPDF`.
   */
  static generateReport(
    sections: Array<{
      title: string;
      content?: string;
      table?: { data: Record<string, unknown>[]; columns: FinancialTableColumn[] };
      chart?: SVGElement;
    }>,
    options?: {
      title?: string;
      subtitle?: string;
      watermark?: string;
      includeTOC?: boolean;
      includePageNumbers?: boolean;
    }
  ): jsPDF {
    const doc = new jsPDF();
    const tocEntries: TOCEntry[] = [];
    let pageNum = 1;
 
    // Title page
    if (options?.title) {
      doc.setFontSize(28);
      doc.text(options.title, 20, 60);
      if (options.subtitle) {
        doc.setFontSize(16);
        doc.text(options.subtitle, 20, 75);
      }
      doc.setFontSize(12);
      doc.text(`Generated: ${new Date().toLocaleString()}`, 20, 90);
      doc.addPage();
      pageNum++;
    }
 
    // TOC placeholder (fill in after)
    const tocPage = pageNum;
 
    // Content sections
    for (const section of sections) {
      doc.addPage();
      pageNum++;
 
      tocEntries.push({ title: section.title, level: 0, page: pageNum });
      doc.setFontSize(18);
      doc.text(section.title, 20, 30);
      doc.line(20, 33, 190, 33);
 
      let y = 45;
 
      if (section.content) {
        doc.setFontSize(11);
        const lines = doc.splitTextToSize(section.content, 170);
        doc.text(lines, 20, y);
        y += lines.length * 6;
      }
 
      if (section.table) {
        y = this.addFinancialTable(doc, section.table.data, section.table.columns, y + 10);
      }
 
      if (section.chart) {
        this.addChart(doc, section.chart, 20, y + 10, 170, 120);
      }
    }
 
    // Fill in TOC
    if (options?.includeTOC && tocEntries.length > 0) {
      doc.setPage(tocPage);
      this.addTableOfContents(doc, tocEntries);
    }
 
    // Header/Footer
    this.addHeaderFooter(doc, {
      header: { title: options?.title ?? 'FinPlan Pro Report' },
      footer: { showPageNumbers: true, showDate: true },
    });
 
    // Watermark
    if (options?.watermark) {
      this.addWatermark(doc, { text: options.watermark });
    }
 
    return doc;
  }
 
  /**
   * Trigger a browser download of the generated PDF.
   *
   * Thin wrapper around `doc.save` that documents the user-facing export
   * intent and gives a single place to attach download-tracking or
   * analytics hooks in the future.
   *
   * @param doc - The composed jsPDF document to serialize and download.
   * @param filename - The default filename suggested to the browser's
   * download dialog (e.g. `"FinPlan-Report-2026.pdf"`). The `.pdf`
   * extension is added by jsPDF if not present.
   * @returns void
   */
  static exportToPDF(doc: jsPDF, filename: string): void {
    doc.save(filename);
  }
}
