// =============================================================================
// AUTO-SAVE ENGINE — Automatic saving with configurable intervals
// Saves to temp file every N minutes, on blur, before close
// Pure TypeScript, deterministic, testable
// =============================================================================
 
/**
 * Configuration options controlling when and how often auto-save runs.
 *
 * @property intervalMs       Interval in milliseconds between automatic interval-triggered saves.
 * @property saveOnBlur       When `true`, triggerBlurSave() performs a save; when `false`, it is a no-op.
 * @property saveOnClose      When `true`, triggerCloseSave() performs a save; when `false`, it is a no-op.
 * @property maxAutoSaves     Maximum number of auto-save entries retained in memory. Older entries are pruned FIFO.
 * @property tempFileSuffix   File suffix appended to the generated temp path (e.g. `.autosave`).
 */
export interface AutoSaveConfig {
  intervalMs: number;
  saveOnBlur: boolean;
  saveOnClose: boolean;
  maxAutoSaves: number;
  tempFileSuffix: string;
}
 
/**
 * Metadata describing a single auto-save snapshot produced by the engine.
 *
 * @property id          Unique identifier for the snapshot (prefixed with `autosave-`).
 * @property timestamp   ISO-8601 UTC timestamp recorded at the moment the save completed.
 * @property path        Generated temp-file path (suffix from {@link AutoSaveConfig.tempFileSuffix}).
 * @property sizeBytes   Approximate byte size of the saved payload (UTF-16 for strings, byteLength for Uint8Array).
 * @property trigger     What caused this save: scheduled interval, window blur, app close, or explicit manual call.
 */
export interface AutoSaveEntry {
  id: string;
  timestamp: string;
  path: string;
  sizeBytes: number;
  trigger: 'interval' | 'blur' | 'close' | 'manual';
}
 
const DEFAULT_CONFIG: AutoSaveConfig = {
  intervalMs: 30 * 1000, // 30 seconds
  saveOnBlur: true,
  saveOnClose: true,
  maxAutoSaves: 5,
  tempFileSuffix: '.autosave',
};
 
/**
 * Pure, deterministic auto-save engine that periodically invokes a caller-supplied
 * save callback and tracks a bounded history of resulting snapshots.
 *
 * The engine is transport-agnostic: it does not perform any I/O itself. Persistence
 * is delegated to the `saveCallback` provided to {@link AutoSaveEngine.start}, and
 * the engine only records metadata (size, timestamp, trigger) about each successful
 * call into an in-memory ring buffer of size {@link AutoSaveConfig.maxAutoSaves}.
 *
 * Lifecycle:
 *  1. Construct with optional partial config (merged over {@link DEFAULT_CONFIG}).
 *  2. Call {@link AutoSaveEngine.start} to begin the interval timer.
 *  3. Optionally call {@link AutoSaveEngine.triggerBlurSave} / {@link AutoSaveEngine.triggerCloseSave}
 *     / {@link AutoSaveEngine.triggerManualSave} for event-driven saves.
 *  4. Call {@link AutoSaveEngine.stop} to clear the interval timer.
 *
 * @typeParam T  The payload type returned by the save callback. Used only for size estimation
 *               (string length × 2, Uint8Array byteLength, otherwise JSON-stringified length × 2).
 */
export class AutoSaveEngine<T = string> {
  private config: AutoSaveConfig;
  private intervalId: ReturnType<typeof setInterval> | null = null;
  private entries: AutoSaveEntry[] = [];
  private isRunning = false;
  private saveCallback: (() => Promise<T | null>) | null = null;
  private onAutoSaveCallback: ((entry: AutoSaveEntry) => void) | null = null;
 
  /**
   * Create a new auto-save engine.
   *
   * @param config  Partial configuration; any omitted fields fall back to {@link DEFAULT_CONFIG}.
   */
  constructor(config: Partial<AutoSaveConfig> = {}) {
    this.config = { ...DEFAULT_CONFIG, ...config };
  }
 
  // --- Lifecycle ---
 
  /**
   * Start the interval-driven auto-save loop. The provided callback is invoked every
   * {@link AutoSaveConfig.intervalMs} milliseconds until {@link AutoSaveEngine.stop} is called.
   *
   * Calling `start` while the engine is already running is a no-op.
   *
   * @param saveCallback  Async function returning the payload to be auto-saved. Returning `null`
   *                      or `undefined` skips entry creation for that tick.
   */
  start(saveCallback: () => Promise<T | null>): void {
    if (this.isRunning) return;
    this.saveCallback = saveCallback;
    this.isRunning = true;
    this.intervalId = setInterval(() => this.performAutoSave('interval'), this.config.intervalMs);
  }
 
  /**
   * Stop the interval-driven auto-save loop and clear the underlying timer.
   * Recorded entries and the save callback reference are preserved.
   */
  stop(): void {
    if (this.intervalId) {
      clearInterval(this.intervalId);
      this.intervalId = null;
    }
    this.isRunning = false;
  }
 
  /**
   * Report whether the interval timer is currently active.
   *
   * @returns `true` if {@link AutoSaveEngine.start} has been called without a matching {@link AutoSaveEngine.stop}.
   */
  isEngineRunning(): boolean {
    return this.isRunning;
  }
 
  // --- Triggers ---
 
  /**
   * Request a save triggered by a window/element blur event.
   * Returns `null` immediately when {@link AutoSaveConfig.saveOnBlur} is disabled.
   *
   * @returns The resulting {@link AutoSaveEntry}, or `null` if disabled / callback returned no data.
   */
  async triggerBlurSave(): Promise<AutoSaveEntry | null> {
    if (!this.config.saveOnBlur) return null;
    return this.performAutoSave('blur');
  }
 
  /**
   * Request a save triggered by an imminent app/window close event.
   * Returns `null` immediately when {@link AutoSaveConfig.saveOnClose} is disabled.
   *
   * @returns The resulting {@link AutoSaveEntry}, or `null` if disabled / callback returned no data.
   */
  async triggerCloseSave(): Promise<AutoSaveEntry | null> {
    if (!this.config.saveOnClose) return null;
    return this.performAutoSave('close');
  }
 
  /**
   * Request an explicit user-initiated save. Always runs regardless of config flags.
   *
   * @returns The resulting {@link AutoSaveEntry}, or `null` if no save callback is configured
   *          or the callback returned no data.
   */
  async triggerManualSave(): Promise<AutoSaveEntry | null> {
    return this.performAutoSave('manual');
  }
 
  // --- Entry Management ---
 
  /**
   * Get a shallow copy of all retained auto-save entries in insertion order (oldest → newest).
   *
   * @returns A new array; mutating it does not affect engine state.
   */
  getEntries(): AutoSaveEntry[] {
    return [...this.entries];
  }
 
  /**
   * Get the most recently recorded auto-save entry, if any.
   *
   * @returns The newest {@link AutoSaveEntry}, or `null` when no saves have been recorded.
   */
  getLatestEntry(): AutoSaveEntry | null {
    return this.entries.length > 0 ? this.entries[this.entries.length - 1] : null;
  }
 
  /**
   * Remove all recorded auto-save entries from in-memory history.
   * Does not stop the engine or affect external persistence.
   */
  clearEntries(): void {
    this.entries = [];
  }
 
  /**
   * Remove a single auto-save entry by id.
   *
   * @param id  The {@link AutoSaveEntry.id} of the entry to remove.
   * @returns   `true` if a matching entry was found and removed; `false` otherwise.
   */
  removeEntry(id: string): boolean {
    const idx = this.entries.findIndex((e) => e.id === id);
    if (idx === -1) return false;
    this.entries.splice(idx, 1);
    return true;
  }
 
  // --- Config ---
 
  /**
   * Get a defensive copy of the currently active configuration.
   *
   * @returns A new {@link AutoSaveConfig} object reflecting the merged defaults plus any overrides.
   */
  getConfig(): AutoSaveConfig {
    return { ...this.config };
  }
 
  /**
   * Merge the supplied updates into the active configuration. If the engine is currently
   * running, it is automatically restarted so the new interval takes effect immediately.
   *
   * @param updates  Partial config; only the supplied fields are overwritten.
   */
  updateConfig(updates: Partial<AutoSaveConfig>): void {
    this.config = { ...this.config, ...updates };
    if (this.isRunning) {
      this.stop();
      if (this.saveCallback) this.start(this.saveCallback);
    }
  }
 
  // --- Callbacks ---
 
  /**
   * Register an observer invoked once for every successful auto-save (any trigger).
   * Only one observer is supported; subsequent calls replace the previous one.
   *
   * @param callback  Receives the freshly created {@link AutoSaveEntry}.
   */
  onAutoSave(callback: (entry: AutoSaveEntry) => void): void {
    this.onAutoSaveCallback = callback;
  }
 
  // --- Private ---
 
  /**
   * Replace the underlying save callback without restarting the interval timer.
   * Useful when the source of the payload changes after the engine has already started.
   *
   * @param callback  Async function returning the payload to be auto-saved.
   */
  setSaveCallback(callback: () => Promise<T | null>): void {
    this.saveCallback = callback;
  }
 
  private async performAutoSave(trigger: AutoSaveEntry['trigger']): Promise<AutoSaveEntry | null> {
    if (!this.saveCallback) return null;
 
    try {
      const data = await this.saveCallback();
      if (data === null || data === undefined) return null;
 
      const sizeBytes =
        typeof data === 'string'
          ? data.length * 2
          : data instanceof Uint8Array
            ? data.byteLength
            : JSON.stringify(data).length * 2;
 
      const entry: AutoSaveEntry = {
        id: `autosave-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
        timestamp: new Date().toISOString(),
        path: `autosave_${Date.now()}${this.config.tempFileSuffix}`,
        sizeBytes,
        trigger,
      };
 
      this.entries.push(entry);
 
      // Prune old entries
      while (this.entries.length > this.config.maxAutoSaves) {
        this.entries.shift();
      }
 
      this.onAutoSaveCallback?.(entry);
      return entry;
    } catch {
      return null;
    }
  }
}
