/**
 * Assumption Engine — Track financial model assumptions separately from calculations
 * Supports versioning, locking, impact analysis, and history
 */
 
export interface Assumption {
  id: string;
  name: string;
  value: number;
  unit: 'percent' | 'currency' | 'ratio' | 'count';
  category: 'revenue' | 'cost' | 'rate' | 'macro' | 'operational';
  source: string;
  effectiveFrom: string;
  effectiveTo: string;
  lockedBy?: string;
  version: number;
  history: AssumptionHistoryEntry[];
}
 
export interface AssumptionHistoryEntry {
  value: number;
  changedBy: string;
  changedAt: string;
  reason: string;
}
 
export interface ImpactResult {
  assumptionId: string;
  oldValue: number;
  newValue: number;
  delta: number;
  affectedCells: string[];
  estimatedImpact: number;
}
 
export class AssumptionEngine {
  private static assumptions = new Map<string, Assumption>();
 
  /**
   * Create a new assumption and store it in the registry.
   * Generates a unique id and initializes the version counter and history.
   * @param data - The assumption fields without id, version, or history.
   * @returns The newly created Assumption.
   */
  static create(data: Omit<Assumption, 'id' | 'version' | 'history'>): Assumption {
    const id = `asm_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`;
    const assumption: Assumption = {
      ...data,
      id,
      version: 1,
      history: [],
    };
    this.assumptions.set(id, assumption);
    return assumption;
  }
 
  /**
   * Look up a single assumption by id.
   * @param id - The unique assumption identifier.
   * @returns The matching Assumption, or undefined if not present.
   */
  static get(id: string): Assumption | undefined {
    return this.assumptions.get(id);
  }
 
  /**
   * Retrieve every assumption currently registered.
   * @returns Array of all Assumption entries.
   */
  static getAll(): Assumption[] {
    return Array.from(this.assumptions.values());
  }
 
  /**
   * Update the value of an existing assumption, appending a history entry.
   * Increments the version and refuses to mutate assumptions locked by another user.
   * @param id - The assumption id to update.
   * @param value - The new numeric value to apply.
   * @param userId - The id of the user performing the update.
   * @param reason - Human-readable justification for the change.
   * @returns The updated Assumption.
   * @throws Error when the assumption is missing or locked by a different user.
   */
  static update(id: string, value: number, userId: string, reason: string): Assumption {
    const existing = this.assumptions.get(id);
    if (!existing) throw new Error(`Assumption ${id} not found`);
    if (existing.lockedBy && existing.lockedBy !== userId) {
      throw new Error(`Assumption ${id} is locked by ${existing.lockedBy}`);
    }
 
    existing.history.push({
      value: existing.value,
      changedBy: userId,
      changedAt: new Date().toISOString(),
      reason,
    });
 
    existing.value = value;
    existing.version += 1;
    return existing;
  }
 
  /**
   * Lock an assumption so only the locking user can mutate it.
   * @param id - The assumption id to lock.
   * @param userId - The id of the user acquiring the lock.
   * @throws Error when the assumption does not exist.
   */
  static lock(id: string, userId: string): void {
    const assumption = this.assumptions.get(id);
    if (!assumption) throw new Error(`Assumption ${id} not found`);
    assumption.lockedBy = userId;
  }
 
  /**
   * Release a previously held lock on an assumption.
   * Only the user that acquired the lock is allowed to release it.
   * @param id - The assumption id to unlock.
   * @param userId - The id of the user attempting to release the lock.
   * @throws Error when the assumption is missing or held by a different user.
   */
  static unlock(id: string, userId: string): void {
    const assumption = this.assumptions.get(id);
    if (!assumption) throw new Error(`Assumption ${id} not found`);
    if (assumption.lockedBy !== userId) {
      throw new Error(`Only ${assumption.lockedBy} can unlock this assumption`);
    }
    assumption.lockedBy = undefined;
  }
 
  /**
   * Return the full change history for an assumption.
   * @param id - The assumption id to inspect.
   * @returns Array of AssumptionHistoryEntry in chronological order; empty when missing.
   */
  static getHistory(id: string): AssumptionHistoryEntry[] {
    const assumption = this.assumptions.get(id);
    return assumption?.history ?? [];
  }
 
  /**
   * Filter and return all assumptions that match the given category.
   * @param category - The category to match (revenue, cost, rate, macro, operational).
   * @returns Array of matching Assumption entries (empty when none match).
   */
  static getByCategory(category: string): Assumption[] {
    const results: Assumption[] = [];
    for (const assumption of this.assumptions.values()) {
      if (assumption.category === category) {
        results.push(assumption);
      }
    }
    return results;
  }
 
  /**
   * Estimate the impact of changing an assumption to a hypothetical new value.
   * Percent-typed values are scaled to a decimal before computing estimated impact.
   * @param assumptionId - The assumption id to evaluate against.
   * @param newValue - The hypothetical value to test.
   * @returns ImpactResult with the delta and scaled estimated impact.
   * @throws Error when the assumption id is unknown.
   */
  static impactAnalysis(assumptionId: string, newValue: number): ImpactResult {
    const assumption = this.assumptions.get(assumptionId);
    if (!assumption) throw new Error(`Assumption ${assumptionId} not found`);
 
    const delta = newValue - assumption.value;
    const affectedCells: string[] = [];
    const estimatedImpact = delta * (assumption.unit === 'percent' ? 0.01 : 1);
 
    return {
      assumptionId,
      oldValue: assumption.value,
      newValue,
      delta,
      affectedCells,
      estimatedImpact,
    };
  }
 
  /**
   * Remove a single assumption from the registry.
   * @param id - The assumption id to delete.
   * @returns True when the assumption existed and was removed; false otherwise.
   */
  static delete(id: string): boolean {
    return this.assumptions.delete(id);
  }
 
  /**
   * Remove every assumption from the in-memory registry.
   * Intended for test isolation or full reset scenarios.
   */
  static clear(): void {
    this.assumptions.clear();
  }
}
