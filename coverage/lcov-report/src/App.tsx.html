/**
 * Domain-specific RouteGroupWrapper with RouteGroupErrorBoundary + Skeleton Suspense fallback.
 * Each route group gets domain-appropriate error messaging and loading skeleton.
 */
function RouteGroupWrapper({ domain }: { domain: 'core' | 'dataGL' | 'finops' | 'cash' | 'reports' | 'industry' | 'utility' }) {
  return (
    <RouteGroupErrorBoundary domain={domain}>
      <Suspense fallback={<RouteSkeleton domain={domain} />}>
        <Outlet />
      </Suspense>
    </RouteGroupErrorBoundary>
  );
}
